"""
E11 — Brier Score Decomposition (Murphy Decomposition)
Phase 1: CPU-only, runs on existing calibration_raw.csv

Decomposes: Brier = Reliability - Resolution + Uncertainty
- Reliability: how far bin-mean-confidence is from bin-mean-outcome (lower = better calibrated)
- Resolution: how far bin-mean-outcome deviates from global base rate (higher = model distinguishes easy/hard)
- Uncertainty: base rate variance, same for all models (constant)

Key finding to look for:
  If CoT improves Reliability (smaller) without improving Resolution at sub-3B,
  that's the "formatting effect" — CoT teaches the model to express uncertainty
  more carefully without actually making it better at distinguishing easy vs hard.

Run: python e11_brier_decomposition.py
"""

import pandas as pd
import numpy as np
from pathlib import Path

RESULTS = Path("results")

# Model sizes for sorting and sub-3B vs >=3B analysis
MODEL_SIZES = {
    "Qwen2.5-0.5B": 0.5, "Llama-3.2-1B": 1.0, "Qwen2.5-1.5B": 1.5,
    "Gemma-2-2B": 2.0, "Qwen2.5-3B": 3.0, "Llama-3.2-3B": 3.0,
    "Llama-3.1-8B": 8.0,
}


def murphy_decomposition(confidences, outcomes, n_bins=10):
    """
    Murphy decomposition of the Brier score.

    Args:
        confidences: array of confidence values in [0, 1]
        outcomes: array of binary outcomes (0 or 1)
        n_bins: number of equal-width bins

    Returns:
        dict with brier, reliability, resolution, uncertainty
    """
    confs = np.array(confidences, dtype=float)
    outs = np.array(outcomes, dtype=float)
    n = len(confs)
    o_bar = outs.mean()  # global base rate

    bin_edges = np.linspace(0, 1, n_bins + 1)
    reliability, resolution = 0.0, 0.0

    bin_details = []
    for i in range(n_bins):
        if i < n_bins - 1:
            mask = (confs >= bin_edges[i]) & (confs < bin_edges[i + 1])
        else:
            # Include right edge in last bin
            mask = (confs >= bin_edges[i]) & (confs <= bin_edges[i + 1])

        n_k = mask.sum()
        if n_k == 0:
            continue

        conf_k = confs[mask].mean()
        out_k = outs[mask].mean()
        reliability += (n_k / n) * (conf_k - out_k) ** 2
        resolution += (n_k / n) * (out_k - o_bar) ** 2

        bin_details.append({
            "bin": f"[{bin_edges[i]:.1f}, {bin_edges[i+1]:.1f}]",
            "n": int(n_k),
            "mean_conf": round(conf_k, 4),
            "mean_acc": round(out_k, 4),
            "gap": round(abs(conf_k - out_k), 4),
        })

    uncertainty = o_bar * (1 - o_bar)
    brier = reliability - resolution + uncertainty

    return {
        "brier": round(brier, 4),
        "reliability": round(reliability, 4),
        "resolution": round(resolution, 4),
        "uncertainty": round(uncertainty, 4),
        "base_rate": round(o_bar, 4),
        "bin_details": bin_details,
    }


def main():
    # --- Locate calibration CSV ---
    cal_csv = RESULTS / "calibration_raw.csv"
    if not cal_csv.exists():
        candidates = list(RESULTS.glob("*calib*.csv"))
        all_csvs = list(RESULTS.glob("*.csv"))
        print(f"calibration_raw.csv not found.")
        print(f"Available CSVs: {[c.name for c in all_csvs]}")
        if candidates:
            cal_csv = candidates[0]
            print(f"Using fallback: {cal_csv.name}")
        else:
            raise FileNotFoundError(
                "No calibration CSV found in results/. "
                "Run the main pipeline (Steps 1-3) first to generate calibration data."
            )

    df = pd.read_csv(cal_csv)
    print(f"Loaded {len(df)} rows from {cal_csv.name}")
    print(f"Columns: {list(df.columns)}")

    # --- Validate expected columns ---
    # Schema: model, size_b, mode, dataset, id, answer, predicted, confidence, correct, raw_response
    # confidence is already [0, 1] (parse_confidence normalizes)
    required = ["model", "mode", "confidence", "correct"]
    missing = [c for c in required if c not in df.columns]
    if missing:
        raise ValueError(f"Missing required columns: {missing}")

    # --- Decompose per (model, mode, dataset) ---
    group_cols = ["model", "mode"]
    if "dataset" in df.columns:
        group_cols.append("dataset")

    results = []
    for keys, group in df.groupby(group_cols):
        if len(group_cols) == 3:
            model, mode, dataset = keys
        else:
            model, mode = keys
            dataset = "all"

        # Drop rows with missing confidence or correctness
        valid = group.dropna(subset=["confidence", "correct"])
        if len(valid) < 10:
            print(f"  Skipping {model}/{mode}/{dataset}: only {len(valid)} valid rows")
            continue

        decomp = murphy_decomposition(valid["confidence"], valid["correct"])
        row = {
            "model": model,
            "mode": mode,
            "dataset": dataset,
            "size_b": MODEL_SIZES.get(model, None),
            "n": len(valid),
            "brier": decomp["brier"],
            "reliability": decomp["reliability"],
            "resolution": decomp["resolution"],
            "uncertainty": decomp["uncertainty"],
            "base_rate": decomp["base_rate"],
        }
        results.append(row)

    out = pd.DataFrame(results)

    # Sort by model size then mode
    out["_sort"] = out["model"].map(MODEL_SIZES).fillna(99)
    out = out.sort_values(["_sort", "mode", "dataset"]).drop(columns=["_sort"])

    print("\n" + "=" * 80)
    print("BRIER DECOMPOSITION: Brier = Reliability - Resolution + Uncertainty")
    print("  Reliability ↓ = better calibrated bins")
    print("  Resolution  ↑ = better at distinguishing easy vs hard questions")
    print("=" * 80 + "\n")
    print(out[["model", "mode", "dataset", "n", "brier", "reliability", "resolution", "uncertainty"]].to_string(index=False))

    # --- Save full results ---
    out_path = RESULTS / "e11_brier_decomposition.csv"
    out.to_csv(out_path, index=False)
    print(f"\nSaved: {out_path}")

    # --- Compute CoT deltas for each model ---
    print("\n" + "=" * 80)
    print("CoT EFFECT ON BRIER COMPONENTS (CoT minus Direct)")
    print("  Reliability Δ < 0 means CoT improved calibration of bins")
    print("  Resolution  Δ > 0 means CoT improved discrimination ability")
    print("=" * 80 + "\n")

    delta_rows = []
    for model in out["model"].unique():
        for dataset in out["dataset"].unique():
            direct = out[(out["model"] == model) & (out["mode"] == "direct") & (out["dataset"] == dataset)]
            cot = out[(out["model"] == model) & (out["mode"] == "cot") & (out["dataset"] == dataset)]
            if direct.empty or cot.empty:
                continue

            d, c = direct.iloc[0], cot.iloc[0]
            size = MODEL_SIZES.get(model, "?")
            scale = "sub-3B" if isinstance(size, (int, float)) and size < 3 else "≥3B"

            rel_delta = c["reliability"] - d["reliability"]
            res_delta = c["resolution"] - d["resolution"]

            print(f"{model} ({size}B, {scale}) — {dataset}:")
            print(f"  Reliability: {d['reliability']:.4f} → {c['reliability']:.4f}  (Δ={rel_delta:+.4f}) {'✓ improved' if rel_delta < 0 else '✗ worsened'}")
            print(f"  Resolution:  {d['resolution']:.4f} → {c['resolution']:.4f}  (Δ={res_delta:+.4f}) {'✓ improved' if res_delta > 0 else '✗ worsened'}")
            print()

            delta_rows.append({
                "model": model, "dataset": dataset, "size_b": size, "scale": scale,
                "reliability_delta": round(rel_delta, 4),
                "resolution_delta": round(res_delta, 4),
                "brier_delta": round(c["brier"] - d["brier"], 4),
            })

    if delta_rows:
        delta_df = pd.DataFrame(delta_rows)
        delta_path = RESULTS / "e11_brier_deltas.csv"
        delta_df.to_csv(delta_path, index=False)
        print(f"Saved deltas: {delta_path}")

        # --- Summary: sub-3B vs >=3B average ---
        print("\n" + "=" * 80)
        print("SUMMARY: Average CoT effect by scale group")
        print("=" * 80 + "\n")
        for scale in ["sub-3B", "≥3B"]:
            sg = delta_df[delta_df["scale"] == scale]
            if sg.empty:
                continue
            print(f"  {scale} models (n={len(sg)}):")
            print(f"    Avg Reliability Δ: {sg['reliability_delta'].mean():+.4f}")
            print(f"    Avg Resolution  Δ: {sg['resolution_delta'].mean():+.4f}")
            print(f"    Avg Brier       Δ: {sg['brier_delta'].mean():+.4f}")
            print()

    print("\n=== INTERPRETATION GUIDE ===")
    print("If sub-3B: Reliability improves (Δ<0) but Resolution doesn't (Δ≤0)")
    print("  → 'Formatting effect': CoT helps express uncertainty, not judge difficulty")
    print("If ≥3B: BOTH improve")
    print("  → 'Computing effect': CoT genuinely reduces uncertainty")


if __name__ == "__main__":
    main()
