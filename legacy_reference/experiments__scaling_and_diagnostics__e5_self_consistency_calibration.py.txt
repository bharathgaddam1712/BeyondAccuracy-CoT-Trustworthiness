"""
E5 — Self-Consistency as a Calibration Signal
Phase 1: CPU-only, runs on existing consistency_raw.csv

Uses the fraction of paraphrase variants that agree on the plurality answer
as an *implicit* confidence score — no verbalization needed.

Key finding to look for:
  If self-consistency ECE < verbalized ECE for small models,
  → practical deployment advice: "Don't ask small models how confident they are,
     just ask them the same question five ways and count agreement."

Run: python e5_self_consistency_calibration.py
"""

import pandas as pd
import numpy as np
from pathlib import Path
from collections import Counter

RESULTS = Path("results")

MODEL_SIZES = {
    "Qwen2.5-0.5B": 0.5, "Llama-3.2-1B": 1.0, "Qwen2.5-1.5B": 1.5,
    "Gemma-2-2B": 2.0, "Qwen2.5-3B": 3.0, "Llama-3.2-3B": 3.0,
    "Llama-3.1-8B": 8.0,
}


def compute_ece(confidences, outcomes, n_bins=10):
    """Standard ECE with equal-width bins over [0, 1]."""
    confs = np.array(confidences, dtype=float)
    outs = np.array(outcomes, dtype=float)
    n = len(confs)
    if n == 0:
        return None

    bin_edges = np.linspace(0, 1, n_bins + 1)
    ece = 0.0
    for i in range(n_bins):
        if i < n_bins - 1:
            mask = (confs >= bin_edges[i]) & (confs < bin_edges[i + 1])
        else:
            mask = (confs >= bin_edges[i]) & (confs <= bin_edges[i + 1])
        if mask.sum() == 0:
            continue
        acc = outs[mask].mean()
        conf = confs[mask].mean()
        ece += (mask.sum() / n) * abs(acc - conf)
    return round(ece, 4)


def main():
    # --- Load consistency results ---
    con_csv = RESULTS / "consistency_raw.csv"
    if not con_csv.exists():
        candidates = list(RESULTS.glob("*consist*.csv"))
        all_csvs = list(RESULTS.glob("*.csv"))
        print(f"consistency_raw.csv not found.")
        print(f"Available CSVs: {[c.name for c in all_csvs]}")
        if candidates:
            con_csv = candidates[0]
            print(f"Using fallback: {con_csv.name}")
        else:
            raise FileNotFoundError(
                "No consistency CSV found in results/. "
                "Run the main pipeline (Steps 1-4) first."
            )

    df = pd.read_csv(con_csv)
    print(f"Loaded {len(df)} rows from {con_csv.name}")
    print(f"Columns: {list(df.columns)}")

    # Schema: model, size_b, mode, dataset, question_id, paraphrase_idx, answer, predicted
    # paraphrase_idx: 0 = original, 1-5 = paraphrases
    required = ["model", "mode", "dataset", "question_id", "answer", "predicted"]
    missing = [c for c in required if c not in df.columns]
    if missing:
        raise ValueError(f"Missing required columns: {missing}")

    # --- Build self-consistency confidence per question ---
    # For each (model, mode, dataset, question_id):
    #   - Collect all predicted answers across paraphrase variants
    #   - Plurality answer = most common prediction
    #   - SC confidence = count(plurality) / total variants
    #   - Correctness = whether plurality answer matches ground truth

    group_cols = ["model", "mode", "dataset", "question_id"]
    sc_rows = []

    for keys, group in df.groupby(group_cols):
        model, mode, dataset, qid = keys

        # Get all non-null predictions
        predictions = group["predicted"].dropna().tolist()
        if len(predictions) < 2:
            continue

        # Normalize predictions for comparison
        preds_normalized = [str(p).strip().lower() for p in predictions]
        counts = Counter(preds_normalized)
        plurality_answer, plurality_count = counts.most_common(1)[0]
        sc_confidence = plurality_count / len(preds_normalized)

        # Check correctness: does plurality match ground truth?
        gold = str(group["answer"].iloc[0]).strip().lower()
        is_correct = int(plurality_answer == gold)

        sc_rows.append({
            "model": model,
            "mode": mode,
            "dataset": dataset,
            "question_id": qid,
            "size_b": MODEL_SIZES.get(model),
            "sc_confidence": round(sc_confidence, 4),
            "is_correct": is_correct,
            "n_variants": len(preds_normalized),
            "n_unique": len(counts),
            "plurality_answer": plurality_answer,
        })

    sc_df = pd.DataFrame(sc_rows)
    print(f"\nBuilt {len(sc_df)} self-consistency records")

    # --- Compute SC-ECE per (model, mode) ---
    print("\n" + "=" * 80)
    print("SELF-CONSISTENCY ECE (using paraphrase agreement as confidence)")
    print("=" * 80 + "\n")
    print(f"{'Model':20s} | {'Mode':6s} | {'SC-ECE':>7s} | {'Mean SC-Conf':>12s} | {'Accuracy':>8s} | {'n':>5s}")
    print("-" * 70)

    sc_ece_rows = []
    for (model, mode), g in sc_df.groupby(["model", "mode"]):
        ece = compute_ece(g["sc_confidence"], g["is_correct"])
        if ece is None:
            continue
        mean_conf = g["sc_confidence"].mean()
        accuracy = g["is_correct"].mean()

        print(f"{model:20s} | {mode:6s} | {ece:7.4f} | {mean_conf:12.4f} | {accuracy:8.4f} | {len(g):5d}")

        sc_ece_rows.append({
            "model": model,
            "mode": mode,
            "size_b": MODEL_SIZES.get(model),
            "sc_ece": ece,
            "mean_sc_confidence": round(mean_conf, 4),
            "accuracy": round(accuracy, 4),
            "n": len(g),
        })

    sc_ece_df = pd.DataFrame(sc_ece_rows)

    # --- Compare with verbalized ECE if available ---
    cal_ece_csv = RESULTS / "calibration_ece.csv"
    if cal_ece_csv.exists():
        cal_ece = pd.read_csv(cal_ece_csv)
        print("\n" + "=" * 80)
        print("COMPARISON: Self-Consistency ECE vs Verbalized ECE")
        print("=" * 80 + "\n")
        print(f"{'Model':20s} | {'Mode':6s} | {'SC-ECE':>7s} | {'Verb-ECE':>8s} | {'SC better?':>11s} | {'Scale':>6s}")
        print("-" * 75)

        comparison_rows = []
        for _, sc_row in sc_ece_df.iterrows():
            model, mode = sc_row["model"], sc_row["mode"]
            # Match against verbalized ECE (aggregate across datasets)
            verb_matches = cal_ece[(cal_ece["model"] == model) & (cal_ece["mode"] == mode)]
            if verb_matches.empty:
                continue

            verb_ece_avg = verb_matches["ece"].mean()
            sc_ece_val = sc_row["sc_ece"]
            size = MODEL_SIZES.get(model, 0)
            scale = "sub-3B" if size < 3 else "≥3B"
            better = "✓ YES" if sc_ece_val < verb_ece_avg else "✗ NO"

            print(f"{model:20s} | {mode:6s} | {sc_ece_val:7.4f} | {verb_ece_avg:8.4f} | {better:>11s} | {scale:>6s}")

            comparison_rows.append({
                "model": model, "mode": mode, "size_b": size, "scale": scale,
                "sc_ece": sc_ece_val, "verb_ece": round(verb_ece_avg, 4),
                "sc_better": sc_ece_val < verb_ece_avg,
            })

        if comparison_rows:
            comp_df = pd.DataFrame(comparison_rows)
            comp_path = RESULTS / "e5_sc_vs_verbalized_comparison.csv"
            comp_df.to_csv(comp_path, index=False)
            print(f"\nSaved comparison: {comp_path}")

            # Summary by scale
            print("\n--- Summary by scale ---")
            for scale in ["sub-3B", "≥3B"]:
                sg = comp_df[comp_df["scale"] == scale]
                if sg.empty:
                    continue
                sc_wins = sg["sc_better"].sum()
                print(f"  {scale}: SC-ECE better in {sc_wins}/{len(sg)} conditions")
    else:
        print(f"\n(calibration_ece.csv not found — skipping verbalized ECE comparison)")

    # --- Save ---
    sc_df.to_csv(RESULTS / "e5_self_consistency.csv", index=False)
    sc_ece_df.to_csv(RESULTS / "e5_self_consistency_ece.csv", index=False)
    print(f"\nSaved: {RESULTS / 'e5_self_consistency.csv'}")
    print(f"Saved: {RESULTS / 'e5_self_consistency_ece.csv'}")

    print("\n=== INTERPRETATION GUIDE ===")
    print("If SC-ECE < Verbalized ECE for sub-3B models:")
    print("  → Practical finding: paraphrase agreement is a better confidence signal")
    print("  → Deployment advice: ask small models 5 ways and count agreement")
    print("If SC-ECE ≈ Verbalized ECE everywhere:")
    print("  → Both signals capture similar information about uncertainty")


if __name__ == "__main__":
    main()
