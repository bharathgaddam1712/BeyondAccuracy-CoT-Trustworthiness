"""
e2_trace_quality_analysis.py
============================
Experiment E2: CoT Trace Quality, Faithfulness, & Degeneracy Analysis
Analyzes 5,600+ generated reasoning traces to determine whether small models
produce structured logic or decorative pseudo-reasoning.

Metrics computed per model:
  1. Mean reasoning step count (using newline & step marker heuristics)
  2. Degeneracy / 4-gram token repetition rate (looping indicator)
  3. Reasoning length & character density
  4. Trace Faithfulness correlation: correlation between trace length and correctness
"""

import re
import numpy as np
import pandas as pd
from pathlib import Path
from scipy.stats import pearsonr

RESULTS = Path("results")
FIGURES = Path("figures")

df = pd.read_csv(RESULTS / "calibration_raw.csv")
cot_df = df[df["mode"] == "cot"].dropna(subset=["raw_response", "correct"]).copy()

print("=" * 95)
print(f"EXPERIMENT E2: CoT TRACE QUALITY & FAITHFULNESS ANALYSIS (N={len(cot_df)} traces across 7 models)")
print("=" * 95)

def count_steps(text):
    if not isinstance(text, str):
        return 0
    lines = [l.strip() for l in text.split("\n") if len(l.strip()) > 5]
    step_markers = re.findall(r'(?:step\s*\d+|first|second|then|next|finally|therefore|so,)', text, re.IGNORECASE)
    return max(len(lines), len(step_markers), 1)

def repetition_rate(text, n=4):
    if not isinstance(text, str) or len(text.split()) < n:
        return 0.0
    words = text.lower().split()
    ngrams = [tuple(words[i:i+n]) for i in range(len(words)-n+1)]
    if not ngrams:
        return 0.0
    unique = len(set(ngrams))
    return round(1.0 - (unique / len(ngrams)), 4)

cot_df["step_count"] = cot_df["raw_response"].apply(count_steps)
cot_df["repetition_rate"] = cot_df["raw_response"].apply(repetition_rate)
cot_df["trace_length"] = cot_df["raw_response"].str.len()

summary_rows = []
for model, g in cot_df.groupby("model"):
    size_b = g["size_b"].iloc[0]
    mean_steps = g["step_count"].mean()
    mean_rep = g["repetition_rate"].mean()
    mean_len = g["trace_length"].mean()
    acc = g["correct"].mean()

    corr_len_acc, p_corr = pearsonr(g["trace_length"], g["correct"].astype(int))

    summary_rows.append({
        "model": model,
        "size_b": size_b,
        "accuracy": round(acc, 4),
        "mean_steps": round(mean_steps, 2),
        "mean_rep_rate": round(mean_rep, 4),
        "mean_trace_length": round(mean_len, 1),
        "corr_len_accuracy": round(corr_len_acc, 3),
        "p_value": round(p_corr, 4),
    })

summary_df = pd.DataFrame(summary_rows).sort_values("size_b")
summary_df.to_csv(RESULTS / "e2_trace_quality_summary.csv", index=False)

print(f"{'Model':15s} | {'Size':>4s} | {'Accuracy':>8s} | {'Avg Steps':>9s} | {'Repetition Rate':>16s} | {'Corr(Length, Acc)':>18s} | {'p-value':>8s}")
print("-" * 95)
for _, r in summary_df.iterrows():
    print(f"{r['model']:15s} | {r['size_b']:4.1f} | {r['accuracy']*100:7.1f}% | {r['mean_steps']:9.2f} | {r['mean_rep_rate']:16.4f} | {r['corr_len_accuracy']:+18.3f} | {r['p_value']:8.4f}")

print("\n" + "=" * 95)
print("E2 ANALYSIS COMPLETE — Saved to results/e2_trace_quality_summary.csv")
print("=" * 95)
