"""
extract_v2.py — MULTI-DATASET INTERNAL SIGNAL EXTRACTION (SCALE-UP)
===================================================================
Pooled dataset bank: GSM8K (200) + BBH (180) + TruthfulQA (200) + ARC-Challenge (100) = 680 questions.

Features:
  1. Multi-dataset pooling with `dataset` tag stored per row.
  2. Extracts both token positions:
     - H (answer token / sequence end) [L+1, Hdim]
     - H_pre (pre-answer token / prompt end) [L+1, Hdim]
  3. Consistent logit confidence: mean top-1 softmax prob across generated tokens.
  4. Verbalized confidence and P(True) probe extraction.
  5. Memory-safe adaptive batching + TF32 + SDPA on RTX 4060.
  6. Skips already extracted files and saves compressed .npz.
"""

import argparse
import gc
import re
import sys
from pathlib import Path

import numpy as np
import pandas as pd
import torch
import torch.nn.functional as F
from transformers import AutoModelForCausalLM, AutoTokenizer

# Enable TF32 on RTX 40-series Tensor Cores
torch.backends.cuda.matmul.allow_tf32 = True
torch.backends.cudnn.allow_tf32 = True

ROOT = Path(__file__).resolve().parents[2]
sys.path.insert(0, str(ROOT))

from src.datasets import load_all_datasets
from src.inference import parse_dataset_answer, parse_confidence, normalize_answer

RESULTS = ROOT / "results"
ACTIVATIONS = RESULTS / "activations"
ACTIVATIONS.mkdir(parents=True, exist_ok=True)

MODEL_REGISTRY = {
    "Qwen2.5-0.5B":  "Qwen/Qwen2.5-0.5B-Instruct",
    "Llama-3.2-1B":  "meta-llama/Llama-3.2-1B-Instruct",
    "Qwen2.5-1.5B":  "Qwen/Qwen2.5-1.5B-Instruct",
    "Gemma-2-2B":    "google/gemma-2-2b-it",
    "Llama-3.2-3B":  "meta-llama/Llama-3.2-3B-Instruct",
}

SYSTEM_PROMPTS = {
    "direct": (
        "You are a helpful assistant. Provide the final answer directly "
        "with no explanation. End your response with 'Answer: <your answer>'."
    ),
    "cot": (
        "You are a helpful assistant. Think step-by-step to solve the problem. "
        "End your response with 'Answer: <your answer>'."
    ),
    "filler": (
        "You are a helpful assistant. Before answering, share your general thoughts "
        "and associations about the topic. Reflect broadly on what comes to mind, "
        "then give your final answer. End your response with 'Answer: <your answer>'."
    ),
    "scratchpad": (
        "You are a helpful assistant. Work out your reasoning inside <scratchpad>...</scratchpad>, "
        "exploring freely and revising if you notice a mistake. Then close the scratchpad and give:\n"
        "Answer: <your answer>\nConfidence: <0-100>"
    ),
}

MAX_NEW_TOKENS = {"direct": 64, "cot": 384, "filler": 384, "scratchpad": 384}


def free():
    gc.collect()
    if torch.cuda.is_available():
        torch.cuda.empty_cache()


def safe_name(s: str) -> str:
    return re.sub(r"[^A-Za-z0-9_.-]", "_", s)


def build_prompts(tokenizer, questions: list[str], mode: str) -> list[str]:
    sys_msg = SYSTEM_PROMPTS[mode]
    is_gemma = "gemma" in str(getattr(tokenizer, "name_or_path", "")).lower()
    prompts = []
    for q in questions:
        if is_gemma:
            msgs = [
                {"role": "user", "content": f"{sys_msg}\n\nQuestion: {q}\nAnswer:"}
            ]
        else:
            msgs = [
                {"role": "system", "content": sys_msg},
                {"role": "user",   "content": f"Question: {q}\nAnswer:"},
            ]
        prompts.append(
            tokenizer.apply_chat_template(msgs, tokenize=False, add_generation_prompt=True)
        )
    return prompts


def check_correctness(pred: str | None, target: str, dataset: str) -> bool:
    if pred is None:
        return False
    if dataset == "gsm8k":
        try:
            return float(pred.replace(",", "")) == float(target.replace(",", ""))
        except Exception:
            return pred.strip().lower() == target.strip().lower()
    elif dataset in ("truthfulqa", "arc"):
        return pred.strip().upper() == target.strip().upper()
    elif dataset == "bbh":
        return normalize_answer(pred) == normalize_answer(target)
    return pred.strip().lower() == target.strip().lower()


@torch.no_grad()
def compute_p_true_batch(model, tokenizer, questions: list[str], answers: list[str], device: str) -> list[float]:
    """Batched P(True) probe: prompt model if proposed answer is True or False."""
    prompts = [
        f"Question: {q}\nProposed answer: {a}\nIs the proposed answer correct?\nAnswer (True or False):"
        for q, a in zip(questions, answers)
    ]
    inp = tokenizer(prompts, return_tensors="pt", padding=True, truncation=True, max_length=1024).to(device)
    logits = model(**inp).logits[:, -1, :]  # [batch, vocab]

    tid = tokenizer(" True",  add_special_tokens=False).input_ids[0]
    fid = tokenizer(" False", add_special_tokens=False).input_ids[0]

    sub_logits = logits[:, [tid, fid]]
    probs = F.softmax(sub_logits, dim=-1)[:, 0]  # P(True)
    return probs.cpu().tolist()


@torch.no_grad()
def process_batch(model, tokenizer, batch_items: list[dict], mode: str, device: str) -> list[dict]:
    """
    Process a batch of questions:
      1. Generate answers & token-level softmax probabilities.
      2. Single forward pass to extract hidden states (H at last token, H_pre at prompt end).
      3. Compute verbalized confidence, logit confidence, correctness, and P(True).
    """
    max_new = MAX_NEW_TOKENS[mode]
    questions = [item["question"] for item in batch_items]
    batch_size = len(batch_items)

    prompts = build_prompts(tokenizer, questions, mode)
    inp = tokenizer(prompts, return_tensors="pt", padding=True, truncation=True, max_length=1024).to(device)
    padded_prompt_len = inp.input_ids.shape[1]

    # Pass 1: Generate response text and token score logits
    out = model.generate(
        **inp,
        max_new_tokens=max_new,
        do_sample=False,
        output_scores=True,
        return_dict_in_generate=True,
        pad_token_id=tokenizer.eos_token_id,
    )

    gen_sequences = out.sequences  # [batch, padded_prompt_len + gen_len]

    # Decode responses
    answers_text = []
    for i in range(batch_size):
        gen_tokens = gen_sequences[i, padded_prompt_len:]
        decoded = tokenizer.decode(gen_tokens, skip_special_tokens=True).strip()
        answers_text.append(decoded)

    # Extract logit signals: mean top-1 softmax prob across generated tokens
    if out.scores:
        scores_stacked = torch.stack(out.scores, dim=1)  # [batch, gen_len, vocab]
        probs_all = F.softmax(scores_stacked, dim=-1)     # [batch, gen_len, vocab]
        top1_all = probs_all.max(dim=-1).values           # [batch, gen_len]
        top1_means = top1_all.mean(dim=-1).cpu().tolist()
        del scores_stacked, probs_all, top1_all
    else:
        top1_means = [float("nan")] * batch_size

    # Pass 2: Single full forward pass for hidden states
    out_h = model(gen_sequences, output_hidden_states=True, use_cache=False)
    hs = out_h.hidden_states  # tuple of (L+1) tensors: [batch, total_len, Hdim]

    pre_idx = padded_prompt_len - 1
    last_idx = gen_sequences.shape[1] - 1

    h_pre_batch = []
    h_last_batch = []
    for i in range(batch_size):
        h_pre_layers = torch.stack([layer[i, pre_idx, :].cpu() for layer in hs]).float().numpy()
        h_last_layers = torch.stack([layer[i, last_idx, :].cpu() for layer in hs]).float().numpy()
        h_pre_batch.append(h_pre_layers)
        h_last_batch.append(h_last_layers)

    del out, out_h, hs, gen_sequences, inp

    # Pass 3: Batched P(True) probe
    ptrue_list = compute_p_true_batch(model, tokenizer, questions, answers_text, device)

    # Process metrics and parse answers
    results = []
    for i, item in enumerate(batch_items):
        resp = answers_text[i]
        ds = item["dataset"]
        parsed_ans = parse_dataset_answer(resp, ds, mode=mode)
        is_cor = check_correctness(parsed_ans, item["answer"], ds)
        verb_conf = parse_confidence(resp, mode=mode)

        results.append({
            "h_last":  h_last_batch[i],
            "h_pre":   h_pre_batch[i],
            "qid":     item["id"],
            "dataset": ds,
            "y":       int(is_cor),
            "verb":    float(verb_conf) if verb_conf is not None else float("nan"),
            "logit":   float(top1_means[i]),
            "ptrue":   float(ptrue_list[i]),
            "resp":    resp,
        })

    return results


def run_model_pipeline(model_name: str, modes: list[str], all_questions: list[dict], device: str, overwrite: bool = False):
    """Loads model once, processes all modes across pooled questions, and unloads cleanly."""
    hf_id = MODEL_REGISTRY[model_name]

    modes_to_run = []
    for mode in modes:
        out_path = ACTIVATIONS / f"internal_{safe_name(model_name)}_{mode}.npz"
        alt_path = RESULTS / f"internal_{safe_name(model_name)}_{mode}.npz"
        target_path = out_path if out_path.exists() else (alt_path if alt_path.exists() else None)
        
        needs_run = True
        if target_path and not overwrite:
            try:
                data = np.load(target_path, allow_pickle=True)
                if len(data["y"]) >= len(all_questions) and "dataset" in data and "H_pre" in data:
                    needs_run = False
                    print(f"  [Skipping] {target_path.name} already has full scale-up data ({len(data['y'])} questions).", flush=True)
                else:
                    print(f"  [Re-extracting] {target_path.name} has only {len(data['y'])} questions (target: {len(all_questions)}). Upgrading...", flush=True)
            except Exception:
                needs_run = True

        if needs_run:
            modes_to_run.append(mode)

    if not modes_to_run:
        print(f"All modes for {model_name} already have complete scale-up data ({len(all_questions)} questions)!", flush=True)
        return

    print(f"\n" + "=" * 80, flush=True)
    print(f"LOADING MODEL: {model_name} ({hf_id})", flush=True)
    print(f"Modes to extract: {modes_to_run} | Questions: {len(all_questions)}", flush=True)
    print("=" * 80, flush=True)

    is_large = any(x in hf_id for x in ["3B", "8B", "Phi-3", "Phi-3.5"])
    tokenizer = AutoTokenizer.from_pretrained(hf_id, trust_remote_code=False)
    tokenizer.padding_side = "left"
    if tokenizer.pad_token is None:
        tokenizer.pad_token = tokenizer.eos_token

    load_kwargs = dict(
        torch_dtype=torch.float16,
        trust_remote_code=False,
        attn_implementation="sdpa" if not is_large else "eager"
    )

    try:
        if is_large:
            from transformers import BitsAndBytesConfig
            bnb = BitsAndBytesConfig(load_in_4bit=True, bnb_4bit_compute_dtype=torch.float16)
            model = AutoModelForCausalLM.from_pretrained(
                hf_id, quantization_config=bnb, trust_remote_code=False,
                attn_implementation="eager", low_cpu_mem_usage=True
            )
            print(f"  [4-bit loaded successfully on {device}]", flush=True)
        else:
            model = AutoModelForCausalLM.from_pretrained(hf_id, **load_kwargs).to(device)
            print(f"  [fp16 + SDPA loaded on {device}]", flush=True)
    except Exception as e:
        print(f"  [Fallback fp16] Loading {model_name} with default settings due to: {e}", flush=True)
        model = AutoModelForCausalLM.from_pretrained(
            hf_id, torch_dtype=torch.float16, trust_remote_code=False
        ).to(device)

    model.eval()

    # Iterate through modes
    for mode in modes_to_run:
        out_path = ACTIVATIONS / f"internal_{safe_name(model_name)}_{mode}.npz"
        batch_size = 4 if mode == "direct" else 2
        if is_large and mode != "direct":
            batch_size = 1

        print(f"\n  >>> Extracting {mode.upper()} mode ({len(all_questions)} questions) in batches of {batch_size}...", flush=True)

        all_extracted = []
        for i in range(0, len(all_questions), batch_size):
            batch_items = all_questions[i:i + batch_size]
            try:
                extracted = process_batch(model, tokenizer, batch_items, mode, device)
                all_extracted.extend(extracted)
            except torch.cuda.OutOfMemoryError:
                print(f"    [OOM on batch {i}] Recovering with single-item processing...", flush=True)
                free()
                for single_item in batch_items:
                    try:
                        extracted = process_batch(model, tokenizer, [single_item], mode, device)
                        all_extracted.extend(extracted)
                    except Exception as ex:
                        print(f"    Error on item {single_item['id']}: {ex}", flush=True)
            except Exception as e:
                print(f"    Error on batch {i}-{i+len(batch_items)}: {e}", flush=True)
                continue

            if i % 20 == 0:
                free()

            if len(all_extracted) % 40 < batch_size or len(all_extracted) >= len(all_questions):
                n_cor = sum(x["y"] for x in all_extracted)
                print(f"    Progress: {len(all_extracted)}/{len(all_questions)} | Correct: {n_cor}/{len(all_extracted)} ({n_cor/len(all_extracted)*100:.1f}%)", flush=True)

        if len(all_extracted) >= 10:
            np.savez_compressed(
                out_path,
                H=np.stack([x["h_last"] for x in all_extracted]),
                H_pre=np.stack([x["h_pre"] for x in all_extracted]),
                y=np.array([x["y"] for x in all_extracted], dtype=int),
                dataset=np.array([x["dataset"] for x in all_extracted], dtype=str),
                verb=np.array([x["verb"] for x in all_extracted], dtype=float),
                logit=np.array([x["logit"] for x in all_extracted], dtype=float),
                ptrue=np.array([x["ptrue"] for x in all_extracted], dtype=float),
                qids=np.array([x["qid"] for x in all_extracted], dtype=str),
            )
            print(f"  [SAVED] {out_path.name} | H shape: {np.stack([x['h_last'] for x in all_extracted]).shape} | n_pos: {sum(x['y'] for x in all_extracted)}", flush=True)

    del model, tokenizer
    free()
    print(f"Finished and freed {model_name} from VRAM.\n", flush=True)


def main():
    parser = argparse.ArgumentParser(description="Multi-dataset internal signal extraction (scale-up)")
    parser.add_argument("--model", default=None, help="Model name to run (default: all)")
    parser.add_argument("--mode",  default=None, help="Mode: direct/cot/filler (default: all)")
    parser.add_argument("--overwrite", action="store_true", help="Overwrite existing npz files")
    args = parser.parse_args()

    device = "cuda" if torch.cuda.is_available() else "cpu"
    print(f"Running multi-dataset extraction on: {device}", flush=True)

    print("Loading pooled question bank (GSM8K + BBH + TruthfulQA + ARC-Challenge)...", flush=True)
    raw_datasets = load_all_datasets(n_gsm8k=200, n_bbh_per_subtask=60, n_truthfulqa=200, n_arc=100, seed=42)
    all_questions = []
    for ds_name, items in raw_datasets.items():
        all_questions.extend(items)
        print(f"  - {ds_name.upper():12s}: {len(items)} questions", flush=True)
    print(f"Total pooled questions: {len(all_questions)}", flush=True)

    models = [args.model] if args.model else list(MODEL_REGISTRY.keys())
    modes  = [args.mode]  if args.mode  else ["direct", "cot", "filler", "scratchpad"]

    for m_name in models:
        if m_name not in MODEL_REGISTRY:
            print(f"Unknown model: {m_name}. Skipping.", flush=True)
            continue
        run_model_pipeline(m_name, modes, all_questions, device, overwrite=args.overwrite)

    print("=" * 80, flush=True)
    print("ALL SCALE-UP EXTRACTIONS COMPLETED SUCCESSFULLY!", flush=True)
    print("=" * 80, flush=True)


if __name__ == "__main__":
    main()
